\begin{problem}[۲۰ نمره]{ریاضی، فرمول و اثبات}\label{prob:math}
  برای تابع $f(x)=(x-3)^2+1$، نقطهٔ کمینه را به دست آورید
  و یکتایی آن را با استفاده از تحدب اکید توضیح دهید.
\end{problem}
\begin{solution}
  \Part{رابطه‌های چندخطی و ارجاع}
  مشتق‌های مرتبهٔ اول و دوم به صورت زیر محاسبه می‌شوند:
  \begin{align}
    f'(x)&=2(x-3),\label{eq:derivative}\\
    f''(x)&=2>0.\label{eq:convexity}
  \end{align}
  از رابطهٔ \eqnref{eq:derivative}، تنها نقطهٔ ایستا $x=3$ است؛
  رابطهٔ \eqnref{eq:convexity} نیز محدب بودن اکید تابع را نشان می‌دهد.
  \begin{answerbox}
    \vspace{-3mm}
    \[
      x^*=\argmin_{x\in\R} f(x)=3,\qquad f(x^*)=1.
    \]
    \vspace{-3mm}
  \end{answerbox}

  \Part{قضیه و اثبات}
  \begin{theorem}\label{thm:unique}
    تابع اکیدا محدب روی یک مجموعهٔ محدب، حداکثر یک کمینهٔ سراسری دارد.
  \end{theorem}
  \begin{proof}
    فرض کنید دو نقطهٔ متمایز $u$ و $v$ هر دو کمینه باشند. از تحدب اکید:
    \[
      f\!\left(\frac{u+v}{2}\right)
      <\frac{f(u)+f(v)}{2}=f(u).
    \]
    این نتیجه با کمینه بودن $u$ تناقض دارد.
  \end{proof}
\end{solution}

\begin{problem}[۲۰ نمره]{جبر خطی و حسابان}\label{prob:algebra}
  دستگاه خطی زیر را حل کنید و جواب را با کمینه‌سازی نرم باقی‌مانده مقایسه کنید.
  سپس یک تابع چندضابطه‌ای، نرم بردار پاسخ و یک انتگرال را نمایش دهید.
\end{problem}
\begin{solution}
  \Part{ماتریس و بردار}
  \begin{equation}\label{eq:linear-system}
    \underbrace{\begin{bmatrix}3&1\\1&2\end{bmatrix}}_{\mat{A}}
    \underbrace{\begin{bmatrix}x_1\\x_2\end{bmatrix}}_{\vect{x}}
    =\underbrace{\begin{bmatrix}9\\8\end{bmatrix}}_{\vect{b}},
    \qquad \vect{x}=\begin{bmatrix}2\\3\end{bmatrix}.
  \end{equation}
  دترمینان ماتریس برابر $5$ و ناصفر است؛ بنابراین پاسخ دستگاه یکتا است.
  جای‌گذاری جواب نیز $3(2)+3=9$ و $2+2(3)=8$ را به دست می‌دهد.

  \Part{کمترین مربعات و باقی‌مانده}
  در نمایش بهینه‌سازی، همان جواب از کمینه‌سازی تابع $J$ به دست می‌آید:
  \begin{subequations}\label{eq:least-squares}
    \begin{align}
      J(\vect{x})&=\tfrac12\norm{\mat{A}\vect{x}-\vect{b}}_2^2,\\
      \nabla J(\vect{x})&=\mat{A}^{\mathsf T}
        (\mat{A}\vect{x}-\vect{b})=\vect{0},\\
      \vect{x}^*&=(\mat{A}^{\mathsf T}\mat{A})^{-1}
        \mat{A}^{\mathsf T}\vect{b}.
    \end{align}
  \end{subequations}
  رابطهٔ آخر در صورت استقلال خطی ستون‌های $\mat{A}$ معتبر است.
  برای محاسبهٔ عددی معمولا دستگاه خطی حل می‌شود و وارون ماتریس صریحا تشکیل نمی‌شود.
  \begin{answerbox}
    \[
      \vect{x}^*=\begin{bmatrix}2\\3\end{bmatrix},\qquad
      \norm{\mat{A}\vect{x}^*-\vect{b}}_2=0.
    \]
  \end{answerbox}

  \Part{تابع چندضابطه‌ای و انتگرال}
  \begin{equation}\label{eq:piecewise}
    g(x)=\begin{cases}x^2,&x\geq 0,\\-x,&x<0,\end{cases}
    \qquad \norm{\vect{x}}_2=\sqrt{13},\qquad
    \int_0^1 x^2\dd x=\frac{1}{3}.
  \end{equation}
\end{solution}

\begin{problem}[۲۰ نمره]{کد و گزارش نتیجهٔ اجرا}\label{prob:code}
  روش گرادیان کاهشی را برای تابع $f(x)=(x-3)^2+1$ با گام ثابت
  $\eta=0.2$ و مقدار اولیهٔ $x_0=0$ پیاده‌سازی کنید.
\end{problem}
\begin{solution}
  \Part{خواندن کد از فایل مستقل}
  قاعدهٔ به‌روزرسانی $x_{k+1}=x_k-0.2f'(x_k)$ است.
  کد زیر ده گام را اجرا می‌کند و مقدار نهایی تابع و خطا را می‌نویسد.
  \CodeFile[fontsize=\footnotesize]{Python}{گرادیان کاهشی با گام ثابت}{examples/gradient_descent.py}
  \begin{answerbox}[خروجی نمونه پس از ده گام]
    \begin{latin}\codefont\small
      x = 2.981860\quad f(x) = 1.000329\quad error = 0.018140
    \end{latin}
  \end{answerbox}
  نتیجه با نمودار \ref{fig:convergence} سازگار است. برای تابع درجهٔ دوم حاضر،
  شرط $0<\eta<1$ همگرایی را تضمین می‌کند.

  \Part{درج کد در متن و تغییر زبان}
  مثال مستقل زیر نشان می‌دهد که قواعد هایلایت با انتخاب زبان تغییر می‌کنند.
\begin{codeblock}{C++}{جست‌وجوی خطی}
#include <vector>
int find_index(const std::vector<int>& a, int target) {
    for (int i = 0; i < static_cast<int>(a.size()); ++i)
        if (a[i] == target) return i;
    return -1;  // Not found.
}
\end{codeblock}
  هزینهٔ زمانی این جست‌وجو در بدترین حالت $O(n)$ است.
\end{solution}

\begin{problem}[۲۰ نمره]{بلوک‌دیاگرام و فلوچارت}\label{prob:diagram}
  یک سامانهٔ بازخورد منفی و گردش کار یک الگوریتم تکراری را نمایش دهید.
  جهت پیکان‌ها و شرط توقف باید مشخص باشند.
\end{problem}
\begin{solution}
  \Part{سامانهٔ کنترل با بازخورد منفی}
  \begin{figure}[H]
    \begin{diagramarea}
      \begin{tikzpicture}
        \node (r) at (0,0) {$r(t)$};
        \node[assignment sum] (sum) at (1.3,0) {$\sum$};
        \node[assignment block] (controller) at (3.7,0) {$C(s)$};
        \node[assignment block] (plant) at (7,0) {$G(s)$};
        \coordinate (branch) at (9,0);
        \node (output) at (10.2,0) {$y(t)$};
        \node[assignment block] (sensor) at (5.4,-1.9) {$H(s)$};
        \draw[assignment arrow] (r) -- node[above] {$+$} (sum);
        \draw[assignment arrow] (sum) -- node[above] {$e(t)$} (controller);
        \draw[assignment arrow] (controller) -- node[above] {$u(t)$} (plant);
        \draw[assignment arrow] (plant) -- (output);
        \fill[AssignmentInk] (branch) circle (1.7pt);
        \draw[assignment arrow] (branch) |- (sensor.east);
        \draw[assignment arrow] (sensor.west) -| node[pos=.93,left] {$-$} (sum.south);
      \end{tikzpicture}
    \end{diagramarea}
    \caption{بلوک‌دیاگرام با بازخورد منفی؛ $C$ کنترل‌کننده، $G$ فرایند و $H$ حسگر است.}
    \label{fig:feedback}
  \end{figure}
  با فرض سامانه‌های خطی و شرایط اولیهٔ صفر، تابع تبدیل حلقه‌بسته برابر است با:
  \begin{equation}\label{eq:feedback}
    \frac{Y(s)}{R(s)}=\frac{C(s)G(s)}{1+C(s)G(s)H(s)}.
  \end{equation}

  \Part{فلوچارت با برچسب فارسی}

        \node[assignment block,below=of start] (init) {\rl{مقداردهی اولیه}};
        \node[assignment block,below=of init] (update) {\rl{محاسبهٔ گام جدید}};
        \node[assignment decision,below=of update] (test) {\rl{شرط توقف؟}};
        \node[assignment io,below=of test] (end) {\rl{نمایش نتیجه}};
        \draw[assignment arrow] (start) -- (init);
        \draw[assignment arrow] (init) -- (update);
        \draw[assignment arrow] (update) -- (test);
        \draw[assignment arrow] (test) -- node[right] {\rl{بله}} (end);
        \draw[assignment arrow] (test.west) -- ++(-2.1,0)
          node[pos=.5,above] {\rl{خیر}} |- (update.west);
      \end{tikzpicture}
    \end{diagramarea}
    \caption{تکرار محاسبه تا رسیدن به شرط توقف، مانند آستانهٔ خطا یا سقف تعداد گام‌ها.}
    \label{fig:flowchart}
  \end{figure}
\end{solution}
