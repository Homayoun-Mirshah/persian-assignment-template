% !TeX root = main.tex
% تنظیمات مشترک ظاهر؛ این فایل پیش از documentclass خوانده می‌شود.
% برای هر تمرین معمولاً فقط همین فایل و metadata.tex را تغییر دهید.
% providecommand اجازه می‌دهد یک سند، مقدار دلخواهش را پیش از input تعیین کند.

\providecommand{\AssignmentPersianBoldFont}{Vazirmatn Bold}

% default یعنی همان ریاضی کلاسیک لاتک، بدون تغییر فونت یا کجی.
% برای فونت OpenType ریاضی می‌توانید مثلاً Latin Modern Math بنویسید.

\providecommand{\AssignmentInlineCodeSize}{}
\providecommand{\AssignmentCodeTheme}{darcula}
\providecommand{\AssignmentMatlabTheme}{matlab-light}

% اندازه‌های اجزای سند؛ اعداد زیر بر حسب pt هستند.

\providecommand{\AssignmentCourseSize}{16}
\providecommand{\AssignmentStudentSize}{15}
