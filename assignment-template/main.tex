\begin{problem}[۲۵ نمره]{ریاضی و فرمول‌ها}
  صورت مسئله، داده‌های معلوم و فرض‌های لازم در این قسمت نوشته می‌شود.
\end{problem}
\begin{solution}
  مراحل حل را اینجا بنویسید. نمونه‌ای از یک رابطهٔ شماره‌دار:
  \begin{equation}\label{eq:starter}
    f(x)=x^2,\qquad f'(x)=2x.
  \end{equation}
  با استفاده از رابطهٔ \eqnref{eq:starter} نتیجه به دست می‌آید.
  \begin{answerbox}
    پاسخ نهایی و در صورت نیاز واحد و شرایط اعتبار آن را اینجا بنویسید.
  \end{answerbox}
\end{solution}

    \label{fig:starter}
  \end{figure}
\end{solution}
